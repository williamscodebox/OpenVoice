🎙️ 2. Recorder Component (with waveform)
jsx
// src/components/Recorder/Recorder.jsx
import './Recorder.css';

export default function Recorder({ isRecording, onStop }) {
  return (
    <div className="recorder">
      <div className={`record-circle ${isRecording ? 'active' : ''}`} />
      <p>{isRecording ? "Recording…" : "Ready to record"}</p>

      {isRecording && (
        <button className="stop-btn" onClick={onStop}>
          ⏹ Stop
        </button>
      )}
    </div>
  );
}
🧬 3. Voice Selection Carousel
jsx
// src/components/VoiceCarousel/VoiceCarousel.jsx
import './VoiceCarousel.css';

export default function VoiceCarousel({ voices, onSelect }) {
  return (
    <div className="voice-carousel">
      {voices.map(v => (
        <div key={v.id} className="voice-card" onClick={() => onSelect(v)}>
          <img src={v.avatar} alt={v.name} />
          <h3>{v.name}</h3>
          <p>{v.tagline}</p>
        </div>
      ))}
    </div>
  );
}
